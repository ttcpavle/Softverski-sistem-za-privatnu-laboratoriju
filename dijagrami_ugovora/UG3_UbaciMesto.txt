    create participant SO as so: UbaciMesto

    C->>+SO: opsteIzvrsenjeSO(mesto)

    create participant DB as db: DBBroker
    SO->>DB: new DBBroker()

    alt mesto nije tipa Mesto
        SO-->>C: Response(false, "Nije prosledjen objekat tipa Mesto")
    else tip je ispravan
        SO->>+DB: pamtiSlog(mesto)
        DB-->>-SO: sacuvano
        alt sacuvano == false
            SO-->>C: Response(false, "Greska pri unosu mesta")
        else sacuvano == true
            SO-->>C: Response(true)
        end
    end
    deactivate SO
